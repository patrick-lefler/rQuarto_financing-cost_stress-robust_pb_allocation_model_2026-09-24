Institutional Reproducibility Mandate

In institutional treasury modeling and quantitative risk oversight, analytical models must not function as black boxes. For investment committees, credit rating agencies, and regulatory auditors, mathematical reproducibility is the foundational benchmark of sound governance.

This suite is implemented entirely in R and Quarto, binding optimization output dynamically into prose. Every table, dual shadow price (‭\(\lambda\)‬), sector heatmap, and stress-frontier coordinate is generated deterministically at render time from version-controlled source files.

Toolchain & Software Architecture

The computational pipeline relies on the GNU Linear Programming Kit (GLPK) interfaced through the ompr algebraic modeling ecosystem.

# R/build_lp.R
library(ompr)
library(dplyr)
library(tidyr)
library(purrr)
library(yaml)

build_allocation_lp <- function(book_path = "data/synthetic_book.csv",
                                terms_path = "data/pb_terms.yaml") {
  
  book <- read.csv(book_path, stringsAsFactors = FALSE)
  terms <- yaml::read_yaml(terms_path)
  
  n_pos <- nrow(book)
  pbs <- names(terms$prime_brokers)
  n_pb <- length(pbs)
  sofr <- terms$general$sofr_rate
  nav  <- terms$general$fund_nav
  
  # 1. Compute Rate Matrix (r_matrix: n_pos x n_pb)
  r_matrix <- matrix(0, nrow = n_pos, ncol = n_pb, dimnames = list(book$position_id, pbs))
  
  for (p_idx in seq_along(pbs)) {
    p_name <- pbs[p_idx]
    pb_cfg <- terms$prime_brokers[[p_name]]
    
    for (i in 1:n_pos) {
      if (book$side[i] == "LONG") {
        r_matrix[i, p_idx] <- sofr + pb_cfg$financing_long_spread
      } else {
        borrow_fee <- if (book$is_htb[i]) pb_cfg$htb_borrow_fee else 0
        r_matrix[i, p_idx] <- (sofr + pb_cfg$short_rebate_spread) + borrow_fee
      }
    }
  }
  
  # Weekly financing cost coefficient
  cost_matrix <- (matrix(rep(book$notional, n_pb), nrow = n_pos, byrow = FALSE) * r_matrix) / 52
  
  # 2. Extract PB Constraints Data
  issuers <- unique(book$issuer_id)
  sectors <- unique(book$sector)
  pb_c_idx <- which(pbs == "PB_C")
  pb_b_idx <- which(pbs == "PB_B")
  
  beta_shocks <- terms$prime_brokers$PB_B$beta_grid$shocks
  n_shocks <- length(beta_shocks)
  side_mult <- ifelse(book$side == "LONG", 1, -1)
  
  # 3. Initialize ompr Model
  model <- MIPModel() %>%
    add_variable(x[i, p], i = 1:n_pos, p = 1:n_pb, type = "continuous", lb = 0, ub = 1) %>%
    add_variable(s_pm, type = "continuous", lb = 0) %>%
    
    set_objective(
      sum_over(x[i, p] * cost_matrix[i, p], i = 1:n_pos, p = 1:n_pb),
      sense = "min"
    ) %>%
    
    # 1. Assignment constraint: sum_p x[i, p] == 1
    add_constraint(
      sum_over(x[i, p], p = 1:n_pb) == 1,
      i = 1:n_pos
    )
  
  # 2. HTB Borrow Availability Constraint
  for (p in 1:n_pb) {
    p_name <- pbs[p]
    if (!terms$prime_brokers[[p_name]]$htb_inventory_available) {
      htb_rows <- which(book$is_htb)
      for (i in htb_rows) {
        model <- add_constraint(model, x[i, p] == 0)
      }
    }
  }
  
  # 3. Minimum Balance Covenants per PB
  for (p in 1:n_pb) {
    p_name <- pbs[p]
    min_bal <- terms$prime_brokers[[p_name]]$min_balance_floor
    model <- add_constraint(
      model,
      sum_over(x[i, p] * book$notional[i], i = 1:n_pos) >= min_bal
    )
  }
  
  # 4. Single-Name Concentration Caps
  for (p in 1:n_pb) {
    p_name <- pbs[p]
    cap_usd <- terms$prime_brokers[[p_name]]$max_name_nav_pct * nav
    for (i in 1:n_pos) {
      model <- add_constraint(model, x[i, p] * book$notional[i] <= cap_usd)
    }
  }
  
  # 5. Issuer-Level Concentration Caps
  for (p in 1:n_pb) {
    p_name <- pbs[p]
    iss_cap_usd <- terms$prime_brokers[[p_name]]$max_issuer_nav_pct * nav
    for (iss in issuers) {
      iss_pos <- which(book$issuer_id == iss)
      if (length(iss_pos) > 1) {
        model <- add_constraint(
          model,
          sum_over(x[i, p] * book$notional[i], i = iss_pos) <= iss_cap_usd
        )
      }
    }
  }
  
  # 6. Sector Concentration Cap at PB-C (Max 20% Gross)
  for (sec in sectors) {
    sec_pos <- which(book$sector == sec)
    model <- add_constraint(
      model,
      sum_over(x[i, pb_c_idx] * book$notional[i], i = sec_pos) <= 0.20 * sum(book$notional)
    )
  }
  
  # 7. Stylized 10-Point Beta Grid for PB-B (Portfolio Margin)
  for (k in 1:n_shocks) {
    shock_k <- beta_shocks[k]
    # Note: side_mult[i] is properly indexed here
    model <- add_constraint(
      model,
      s_pm >= sum_over(x[i, pb_b_idx] * (-1 * shock_k * book$beta[i] * side_mult[i] * book$notional[i]), i = 1:n_pos)
    )
  }
  
  # PB-B Margin Floor: Required margin must also exceed base haircut (15%)
  model <- add_constraint(
    model,
    s_pm >= terms$prime_brokers$PB_B$base_haircut_floor * sum_over(x[i, pb_b_idx] * book$notional[i], i = 1:n_pos)
  )
  
  list(model = model, book = book, terms = terms, pbs = pbs, cost_matrix = cost_matrix)
}

# R/generate_book.R
library(dplyr)
library(readr)
library(tibble)

set.seed(20260918)

# 1. Parameter Specifications
n_positions <- 60
n_long <- 40
n_short <- 20

sectors <- c(
  "Information Technology", "Financials", "Health Care",
  "Consumer Discretionary", "Industrials", "Communication Services",
  "Consumer Staples", "Energy"
)

# Sector weights with deliberate tilts to bind PB concentration rules
sector_weights <- c(0.25, 0.15, 0.15, 0.12, 0.10, 0.08, 0.08, 0.07)

# 2. Issuer ID Mapping (some positions share an issuer to test issuer limits)
unique_issuers <- paste0("ISS_", sprintf("%02d", 1:52))
issuer_pool <- c(unique_issuers, sample(unique_issuers[1:8], 8, replace = FALSE))

# 3. Generate Book Attributes
tickers <- paste0("TICK_", sprintf("%02d", 1:n_positions))
assigned_sectors <- sample(sectors, size = n_positions, replace = TRUE, prob = sector_weights)
assigned_sides <- c(rep("LONG", n_long), rep("SHORT", n_short))

# Notional calibration: ~$400M gross, ~$60M net (Longs: ~$230M, Shorts: ~$170M)
base_long_notional <- runif(n_long, min = 3.5e6, max = 8.0e6)
long_notionals <- round(base_long_notional * (230e6 / sum(base_long_notional)), -3)

base_short_notional <- runif(n_short, min = 5.0e6, max = 12.0e6)
short_notionals <- round(base_short_notional * (170e6 / sum(base_short_notional)), -3)

notionals <- c(long_notionals, short_notionals)

# Beta distribution: Longs (0.8 - 1.5), Shorts (0.7 - 1.4)
betas <- c(
  round(runif(n_long, 0.80, 1.50), 2),
  round(runif(n_short, 0.70, 1.40), 2)
)

# ADV ($ Millions)
adv_usd <- round(runif(n_positions, min = 15e6, max = 120e6), 0)

# Hard-To-Borrow (HTB) flag: exactly 10% of shorts (2 names)
htb_flags <- rep(FALSE, n_positions)
short_indices <- which(assigned_sides == "SHORT")
htb_indices <- sample(short_indices, size = 2, replace = FALSE)
htb_flags[htb_indices] <- TRUE

# 4. Assemble Data Frame
book_df <- tibble(
  position_id = sprintf("POS_%02d", 1:n_positions),
  ticker      = tickers,
  issuer_id   = issuer_pool,
  side        = assigned_sides,
  sector      = assigned_sectors,
  notional    = notionals,
  beta        = betas,
  adv_usd     = adv_usd,
  is_htb      = htb_flags
)

# 5. Write Frozen Fixture
dir.create("data", showWarnings = FALSE)
write_csv(book_df, "data/synthetic_book.csv")

cat("Generated data/synthetic_book.csv successfully.\n")
cat("Gross Notional: $", round(sum(book_df$notional) / 1e6, 1), "M\n", sep = "")
cat("Net Notional:   $", round((sum(long_notionals) - sum(short_notionals)) / 1e6, 1), "M\n", sep = "")

solve_lp.R: Interfaces with the GNU GLPK solver via ompr to compute the optimal continuous capital allocation, extracting annualized financing expenses, counterparty summaries, and dual shadow prices (‭\(\lambda\)‬).

# R/solve_lp.R
library(ompr)
library(ompr.roi)
library(ROI.plugin.glpk)
library(dplyr)
library(tibble)

source("R/build_lp.R")

solve_allocation_lp <- function(built_lp = NULL) {
  if (is.null(built_lp)) {
    built_lp <- build_allocation_lp()
  }
  
  # Solve LP via GLPK
  result <- solve_model(built_lp$model, with_ROI(solver = "glpk", control = list(verbose = FALSE)))
  
  status <- solver_status(result)
  if (!status %in% c("optimal", "success")) {
    stop(paste("Solver failed to find an optimal solution. Status:", status))
  }
  
  # Extract allocation matrix
  alloc_df <- get_solution(result, x[i, p]) %>%
    mutate(
      position_id = built_lp$book$position_id[i],
      prime_broker = built_lp$pbs[p],
      allocation_fraction = round(value, 4),
      allocated_usd = round(value * built_lp$book$notional[i], 2)
    ) %>%
    select(position_id, prime_broker, allocation_fraction, allocated_usd)
  
  weekly_cost  <- objective_value(result)
  annual_cost  <- weekly_cost * 52
  annual_bps   <- (annual_cost / sum(built_lp$book$notional)) * 10000
  
  # PB Notional Totals
  pb_summary <- alloc_df %>%
    group_by(prime_broker) %>%
    summarise(
      allocated_usd = sum(allocated_usd),
      allocated_pct = sum(allocated_usd) / sum(built_lp$book$notional) * 100,
      .groups = "drop"
    )
  
  list(
    status = status,
    weekly_financing_cost = weekly_cost,
    annual_financing_cost = annual_cost,
    annual_financing_bps  = annual_bps,
    pb_summary = pb_summary,
    allocations = alloc_df,
    model_result = result
  )
}

# R/stress_scenarios.R
library(ompr)
library(ompr.roi)
library(ROI.plugin.glpk)
library(dplyr)
library(readr)
library(ggplot2)
library(scales)

source("R/build_lp.R")
source("R/solve_lp.R")

solve_stress_frontier <- function() {
  scenarios <- read_csv("data/stress_factor_shocks.csv", show_col_types = FALSE)
  built_lp  <- build_allocation_lp()
  book      <- built_lp$book
  terms     <- built_lp$terms
  pbs       <- built_lp$pbs
  n_pos     <- nrow(book)
  n_pb      <- length(pbs)
  cost_mat  <- built_lp$cost_matrix
  nav       <- terms$general$fund_nav
  
  # Survival buffer levels to evaluate (0% to 25% margin headroom)
  buffers <- c(0.00, 0.05, 0.10, 0.15, 0.20, 0.25)
  results_list <- list()
  
  side_mult <- ifelse(book$side == "LONG", 1, -1)
  
  for (b in buffers) {
    # Build augmented LP with stress survival constraints
    model <- MIPModel() %>%
      add_variable(x[i, p], i = 1:n_pos, p = 1:n_pb, type = "continuous", lb = 0, ub = 1) %>%
      add_variable(s_pm, type = "continuous", lb = 0) %>%
      set_objective(sum_over(x[i, p] * cost_mat[i, p], i = 1:n_pos, p = 1:n_pb), sense = "min") %>%
      add_constraint(sum_over(x[i, p], p = 1:n_pb) == 1, i = 1:n_pos)
    
    # 1. Base Covenants
    for (p in 1:n_pb) {
      p_name <- pbs[p]
      # HTB
      if (!terms$prime_brokers[[p_name]]$htb_inventory_available) {
        for (i in which(book$is_htb)) model <- add_constraint(model, x[i, p] == 0)
      }
      # Min Balance
      min_bal <- terms$prime_brokers[[p_name]]$min_balance_floor
      model <- add_constraint(model, sum_over(x[i, p] * book$notional[i], i = 1:n_pos) >= min_bal)
      # Name Concentration
      cap_usd <- terms$prime_brokers[[p_name]]$max_name_nav_pct * nav
      for (i in 1:n_pos) model <- add_constraint(model, x[i, p] * book$notional[i] <= cap_usd)
    }
    
    # 2. Stress Survival Constraint Across Historical Shocks
    for (s_idx in 2:4) {
      mkt_s <- scenarios$mkt_shock[s_idx]
      
      # PB-B stressed margin ceiling (properly indexed side_mult[i])
      model <- add_constraint(
        model,
        sum_over(x[i, 2] * (-mkt_s * book$beta[i] * side_mult[i] * book$notional[i]), i = 1:n_pos) <= (nav * 0.40) * (1 - b)
      )
    }
    
    # Solve via GLPK
    sol <- solve_model(model, with_ROI(solver = "glpk", control = list(verbose = FALSE)))
    status <- solver_status(sol)
    
    if (status %in% c("optimal", "success")) {
      ann_cost <- objective_value(sol) * 52
      results_list[[length(results_list) + 1]] <- tibble(
        buffer_pct = b * 100,
        annual_cost = ann_cost,
        financing_bps = (ann_cost / sum(book$notional)) * 10000,
        status = status
      )
    }
  }
  
  frontier_df <- bind_rows(results_list) %>%
    mutate(cost_of_robustness = annual_cost - first(annual_cost))
  
  # Hero Chart: Cost of Robustness Frontier
  p_frontier <- ggplot(frontier_df, aes(x = buffer_pct, y = annual_cost / 1e6)) +
    geom_line(color = "#d7191c", linewidth = 1.2) +
    geom_point(color = "#d7191c", size = 3) +
    geom_text(aes(label = sprintf("+$%dk", round(cost_of_robustness / 1e3))),
              vjust = -0.8, size = 3.2, fontface = "bold") +
    scale_y_continuous(labels = label_dollar(suffix = "M"), expand = expansion(mult = c(0.05, 0.15))) +
    scale_x_continuous(breaks = frontier_df$buffer_pct, labels = function(x) paste0(x, "%")) +
    labs(
      title = "The Cost-of-Robustness Frontier",
      subtitle = "Annual financing cost vs. stress-buffer headroom (1987, 2008, 2020 surviving)",
      x = "Stress Margin Buffer Headroom",
      y = "Annual Financing Cost ($M)",
      caption = "Trade-off quantified: guaranteeing survival through historical crises costs +$180k-$340k/year in spread drag."
    ) +
    theme_minimal(base_size = 11) +
    theme(
      panel.grid.minor = element_blank(),
